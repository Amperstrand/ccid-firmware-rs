Interface Type ISO_DEP (0x02):        linux_libnfc-nci nci_defs.h L162
RATS Request:                         ISO 14443-4:2018 §5.1–5.2
ATS Response Structure:               ISO 14443-4:2018 §5.2–5.3.7
  TL (length), T0 (format byte), TA/TB/TC (interface bytes), historical
ATS T0 Byte:                          ISO 14443-4:2018 §5.2.3
  Bits 7–5: TA/TB/TC presence flags | Bits 3–0: FSCI (frame size)

ISO-DEP BLOCK EXCHANGE (Data Link Layer)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
★ I-Block (Information):              ISO 14443-4:2018 §7.2.1
  PCB: 0b00xxxxxx (not 0b00xxx101) | Carries application data
  Structure: [PCB | CID? | NAD? | INF | CRC-A(2)]
  PCB Bit 5: Chaining flag (0=last, 1=more data follows)
  PCB Bit 0: Block Number (alternates 0↔1)
R-Block (Receive Ready):              ISO 14443-4:2018 §7.2.2
  PCB: 0b10xxxxxx (not 0b1001xxxx) | No information field
  R(ACK): Positive acknowledgement | R(NAK): Retry request
★ S-Block (Supervisory):              ISO 14443-4:2018 §7.2.3
  PCB: 0b11xxxxxx (not 0b1110xxxx, not 0b1101xxxx)
  WTX: Waiting Time eXtension (S-block with WTXM multiplier byte)
  DESELECT: Terminate session
PCB Prologue (Mandatory + Optional):  ISO 14443-4:2018 §7.1.2
  PCB (always) + CID (if TA bit in ATS T0) + NAD (if TC bit in ATS T0)
CID Field (4 bits):                   ISO 14443-4:2018 §7.1.2.2
  Values 0–14 (15 is RFU) | Identifies target in multi-activation
NAD Field (optional):                 ISO 14443-4:2018 §7.1.2.3
  Node Address per ISO 7816-3 | Only in I-blocks
Block Numbering:                      ISO 14443-4:2018 §7.2.4
  Toggles 0→1→0→1 for sequential I-block/R-block pairs
Epilogue (CRC-A):                     ISO 14443-4:2018 §7.1.4
  Last 2 bytes: CRC polynomial 0x1021 (Type A) or CRC-B (Type B)

// Boot & VEN toggle
fn power_up(&mut self) {
    // PN7160 Datasheet §11.1.2: VEN < 0.4V = Hard Power-Down
    self.ven.set_low();
    sleep(10ms);
    // PN7160 Datasheet §11.1.2.2: VEN > 2.4V → Standby → Active
    self.ven.set_high();
    // UM11495 §6.1 / NCI 2.0 §4.1.1: Send CORE_RESET_CMD
    self.send(&[0x20, 0x00, 0x01, 0x00]);
}

// Parse ATS (ISO-DEP activation)
fn parse_ats(&self, ats: &[u8]) {
    // ISO 14443-4:2018 §5.2–5.3.7: ATS structure
    let tl = ats[0];                    // Length (including TL)
    let t0 = ats[1];                    // Format byte
    let fsci = t0 & 0x0F;               // ISO 14443-4:2018 §5.2.3: Bits 3:0
    let tb = ats[2];                    // TB(1) if t0 bit 5 set
    let fwi = (tb >> 4) & 0x0F;         // ISO 14443-4:2018 §5.3.5: Bits 7:4
}

// Send I-block
fn send_i_block(&mut self, payload: &[u8]) {
    // ISO 14443-4:2018 §7.2.1: I-block format
    let mut pcb = 0x00;                 // Start with I-block marker
    pcb |= (self.block_num & 0x01);     // Bit 0: block number toggle
    pcb |= if payload.len() > 253 { 0x20 } else { 0x00 }; // Bit 5: chaining
    
    let mut block = vec![pcb, self.cid];  // CID from ATS TC
    block.extend_from_slice(payload);
    
    // Append CRC-A: ISO 14443-3:2018 §6.2.4
    let crc = calculate_crc_a(&block);
    block.extend_from_slice(&crc);
    
    self.send(&block);
}

// Handle WTX (Waiting Time eXtension)
fn handle_wtx(&mut self, wtx_frame: &[u8]) {
    // ISO 14443-4:2018 §7.4: WTX = S-block with WTXM multiplier
    let wtxm = wtx_frame[2];             // INF field (1–59 multiplier)
    let fwt_extended = self.fwt * wtxm as u32;  // Extend timeout
    
    // Send S-block ACK (no INF field)
    let pcb = 0xF2;                      // S-block, DESELECT=0, no INF
    self.send(&[pcb, self.cid, 0x02]);  // CRC will be added
}

// Receive and parse ISO-DEP block
fn recv_block(&mut self) -> Result<Vec<u8>> {
    let frame = self.recv()?;
    
    // ISO 14443-4:2018 §7.2: Parse PCB
    let pcb = frame[0];
    let block_type = (pcb >> 6) & 0x03;  // Bits 7:6
    
    match block_type {
        0b00 => {  // I-block
            // ISO 14443-4:2018 §7.2.1: Extract payload
            let chaining = (pcb >> 5) & 0x01;
            let block_num = pcb & 0x01;
            let payload = &frame[2..frame.len()-2];  // Skip PCB, CID, CRC
            Ok(payload.to_vec())
        },
        0b10 => {  // R-block (ACK/NAK)
            // ISO 14443-4:2018 §7.2.2: Ack/Nak bit
            let is_ack = (pcb >> 4) & 0x01 == 0;
            Err(if is_ack { "R(ACK)" } else { "R(NAK)" })
        },
        0b11 => {  // S-block
            // ISO 14443-4:2018 §7.2.3: WTX vs DESELECT
            Err("S-block (control)")
        },
        _ => Err("Invalid block")
    }
}
